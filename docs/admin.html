→ 형식: A열=번호, B열=매장명, C열=주소, D열=연락처, E열=상세URL, F열=비고

연동 방법 (Google Apps Script)

  1. Google 시트를 새로 만들고 열기
  2. 메뉴 → 확장 프로그램 → Apps Script 클릭
  3. 아래 [스크립트 복사] 버튼으로 코드 붙여넣기
  4. 저장 후 배포 → 새 배포 → 웹 앱으로 배포
  5. 생성된 웹 앱 URL을 아래에 붙여넣기
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  // 헤더가 없으면 추가
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['접수일시','브랜드','지역','상담가능시간','연락처','요청사항','상태','메모']);
  }
  
  sheet.appendRow([
    data.created_at || new Date().toLocaleString('ko-KR'),
    data.brand_name || '',
    data.region || '',
    data.available_time || '',
    data.phone_number || '',
    data.requests || '',
    data.status || '대기중',
    data.memo || ''
  ]);
  
  return ContentService
    .createTextOutput(JSON.stringify({result: 'success'}))
    .setMimeType(ContentService.MimeType.JSON);
}
